\newcommand{\SlateFrameHeader}[3]{%
  \node[anchor=north west,inner sep=0pt,text=BSVPrimary,
        font=\fontsize{16}{19}\selectfont\bfseries]
    at ([xshift=0.76cm,yshift=-0.46cm]current page.north west) {#1};
  \node[anchor=north east,inner sep=0pt,text=BSVMuted,
        font=\fontsize{6.8}{8.5}\selectfont\bfseries]
    at ([xshift=-1.48cm,yshift=-0.59cm]current page.north east) {#2};
  \node[anchor=north east,inner sep=0pt,text=BSVPrimary,
        font=\fontsize{9}{11}\selectfont\bfseries]
    at ([xshift=-0.76cm,yshift=-0.53cm]current page.north east) {#3};
  \draw[BSVLine,line width=0.5pt]
    ([xshift=0.76cm,yshift=-1.16cm]current page.north west) --
    ([xshift=-0.76cm,yshift=-1.16cm]current page.north east);
  \draw[BSVPrimary,line width=1.0pt]
    ([xshift=0.76cm,yshift=-1.16cm]current page.north west) -- ++(1.18cm,0);
}
